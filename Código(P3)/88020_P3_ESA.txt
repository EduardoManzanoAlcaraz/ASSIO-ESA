# [alex] Unidades del grupo creadas en /etc/systemd/system/

#   After=network.target
#
#   [Service]
#   Type=simple
#   User=usuario_a
#   ExecStart=/opt/aurora6/tools/structure-monitor.sh
#   Restart=on-failure
#   RestartSec=3
#
#   [Install]

sudo journalctl -u structure-monitor.service --since "5 minutes ago"


---------------------------------------------------------------------
APARTADO 5. Endurecimiento del servicio y mínimo privilegio
---------------------------------------------------------------------

# Revisión de propietarios y permisos
namei -l /opt/aurora6/tools/structure-monitor.sh
stat -c "%U:%G %a %n" /opt/aurora6/tools/structure-monitor.sh
stat -c "%U:%G %a %n" /etc/systemd/system/structure-monitor.service

# ¿Puede usuario_a modificar el script? ¿Qué privilegios tiene?
test -w /opt/aurora6/tools/structure-monitor.sh && echo "ALERTA: modificable" || echo "OK: sin escritura"
sudo -l

# Edición de la unidad para añadir las medidas de aislamiento
sudo nano /etc/systemd/system/structure-monitor.service

# Líneas añadidas en la sección [Service]:
#   NoNewPrivileges=true
#   PrivateTmp=true
#   ProtectHome=true
#   ProtectSystem=full

# Aplicar los cambios y comprobarlos
sudo systemctl daemon-reload
sudo systemctl restart structure-monitor.service
systemctl show structure-monitor.service -p User -p NoNewPrivileges -p PrivateTmp -p ProtectHome -p ProtectSystem
systemctl is-active structure-monitor.service


---------------------------------------------------------------------
APARTADO 6. Recuperación automática ante un fallo
---------------------------------------------------------------------

systemctl status structure-monitor.service
systemctl show structure-monitor.service -p MainPID -p NRestarts

# Simulación controlada de fallo (cierre forzado del proceso)
sudo systemctl kill --kill-who=main --signal=SIGKILL structure-monitor.service
sleep 5

systemctl status structure-monitor.service
systemctl show structure-monitor.service -p MainPID -p NRestarts
sudo journalctl -u structure-monitor.service --since "2 minutes ago"


---------------------------------------------------------------------
APARTADO 7. Verificación puntual de los registros del servicio
---------------------------------------------------------------------

sudo journalctl -u structure-monitor.service -n 15 --no-pager
